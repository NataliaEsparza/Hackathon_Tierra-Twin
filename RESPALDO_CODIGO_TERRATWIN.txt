// ENLACES DE GOOGLE MAPS
function parseGoogleMapsOrCoords(rawInput) {
  const text = rawInput.trim();

  // 1. Pin exacto dentro de URL de Google Maps (!3dLAT!4dLON)
  const exactPinMatch = text.match(/!3d(-?\d+\.\d+)!4d(-?\d+\.\d+)/);
  // 2. Centro de vista en URL de Google Maps (@LAT,LON)
  const atCoordsMatch = text.match(/@(-?\d+\.\d+),(-?\d+\.\d+)/);
  // 3. Parámetro ?q=LAT,LON o ll=LAT,LON
  const queryCoordsMatch = text.match(/[?&](?:q|ll)=(-?\d+\.\d+)(?:,|%2C)\s*(-?\d+\.\d+)/i);
  // 4. Coordenadas escritas directamente: "24.0277, -104.6532"
  const plainCoordsMatch = text.match(/^(-?\d{1,2}(?:\.\d+)?)\s*,\s*(-?\d{1,3}(?:\.\d+)?)$/);

  const match = exactPinMatch || queryCoordsMatch || atCoordsMatch || plainCoordsMatch;
  if (!match) return null;

  const lat = Number(match[1]);
  const lon = Number(match[2]);
  if (Number.isNaN(lat) || Number.isNaN(lon) || Math.abs(lat) > 90 || Math.abs(lon) > 180) {
    return null;
  }

  // Extraer nombre del lugar si viene en la URL (/place/Nombre+Del+Lugar/)
  let extractedName = '';
  const placeMatch = text.match(/\/place\/([^/@]+)/);
  if (placeMatch && placeMatch[1]) {
    try {
      extractedName = decodeURIComponent(placeMatch[1].replace(/\+/g, ' ')).trim();
    } catch {
      extractedName = placeMatch[1].replace(/\+/g, ' ').trim();
    }
  }

  return { lat, lon, extractedName };
}

async function reverseGeocodeCoords(lat, lon) {
  try {
    const response = await axios.get('https://nominatim.openstreetmap.org/reverse', {
      params: {
        format: 'jsonv2',
        lat: lat.toFixed(5),
        lon: lon.toFixed(5),
        'accept-language': 'es'
      },
      headers: {
        'User-Agent': 'TerraTwin-NASA-SpaceApps/3.1 (educational-project)'
      },
      timeout: 4500
    });

    const addr = response.data?.address || {};
    const specificFeature =
      response.data?.name ||
      addr.natural ||
      addr.volcano ||
      addr.peak ||
      addr.protected_area ||
      addr.desert ||
      '';

    let city =
      addr.city ||
      addr.town ||
      addr.village ||
      addr.municipality ||
      addr.county ||
      addr.region ||
      '';

    if (city.toLowerCase().includes('victoria de durango')) {
      city = 'Durango';
    }

    const state = addr.state || '';
    const country = addr.country || '';

    const parts = [specificFeature, city, state, country].filter(
      (p, idx, arr) => p && arr.indexOf(p) === idx
    );

    if (parts.length > 0) {
      return {
        placeName: parts.slice(0, 3).join(', '),
        regionSubtitle: [state, country].filter(Boolean).join(', ') || 'Coordenada satelital'
      };
    }

    return {
      placeName: `Sector (${lat.toFixed(4)}°, ${lon.toFixed(4)}°)`,
      regionSubtitle: 'Coordenada satelital'
    };
  } catch {
    return {
      placeName: `Sector (${lat.toFixed(4)}°, ${lon.toFixed(4)}°)`,
      regionSubtitle: 'Coordenada satelital'
    };
  }
}

async function fetchNasaPowerByCoords(lat, lon, cacheKey = null, fallback = null) {
  if (cacheKey && nasaClimateCache.has(cacheKey)) {
    return { ...nasaClimateCache.get(cacheKey), source: 'NASA satélite (caché)' };
  }

  try {
    const url = 'https://power.larc.nasa.gov/api/temporal/climatology/point';
    const response = await axios.get(url, {
      params: {
        parameters: 'T2M_MIN,T2M_MAX,ALLSKY_SFC_SW_DWN,PRECTOTCORR',
        community: 'RE',
        longitude: Number(lon.toFixed(4)),
        latitude: Number(lat.toFixed(4)),
        format: 'JSON'
      },
      timeout: 5000
    });

    const params = response.data?.properties?.parameter;
    if (!params || !params.T2M_MIN || params.T2M_MIN.ANN === -999) {
      throw new Error('Datos incompletos de NASA POWER');
    }

    const climateData = {
      tMin: Number(params.T2M_MIN.ANN.toFixed(2)),
      tMax: Number(params.T2M_MAX.ANN.toFixed(2)),
      solar: Number(params.ALLSKY_SFC_SW_DWN.ANN.toFixed(2)),
      precip: Number(params.PRECTOTCORR.ANN.toFixed(2))
    };

    if (cacheKey) {
      nasaClimateCache.set(cacheKey, climateData);
    }
    return { ...climateData, source: 'NASA satélite (en vivo)' };
  } catch {
    if (fallback) {
      return { ...fallback, source: 'Base científica local' };
    }
    const absLat = Math.abs(lat);
    return {
      tMin: Number((22 - absLat * 0.65).toFixed(1)),
      tMax: Number((38 - absLat * 0.45).toFixed(1)),
      solar: Number(clamp(6.8 - absLat * 0.05, 2.1, 7.2).toFixed(2)),
      precip: 1.2,
      source: 'Estimación orbital local'
    };
  }
}

async function fetchElevationByCoords(lat, lon) {
  try {
    const response = await axios.get('https://api.open-elevation.com/api/v1/lookup', {
      params: { locations: `${lat.toFixed(4)},${lon.toFixed(4)}` },
      timeout: 3000
    });
    const elev = response.data?.results?.[0]?.elevation;
    return typeof elev === 'number' ? Math.round(elev) : null;
  } catch {
    return null;
  }
}

function buildEarthNormalizedProfile(site, climate, target) {
  const thermal = clamp(((30 - climate.tMin) / 70) * 100, 0, 100);
  const radiation = clamp(((climate.solar - 2.0) / 5.5) * 100, 0, 100);
  const aridity = clamp(((3.0 - Math.min(climate.precip, 3.0)) / 3.0) * 100, 0, 100);
  const topography = clamp(site.slopeIndex, 0, 100);

  let geology = 45;
  if (site.geologyType === target.requiredGeology) {
    geology = 95;
  } else if (
    (site.geologyType === 'basaltic-volcanic' && target.requiredGeology === 'anorthosite-regolith') ||
    (site.geologyType === 'permafrost-regolith' && target.requiredGeology === 'anorthosite-regolith')
  ) {
    geology = 72;
  }

  let subsurface = 20;
  if (target.requiresCaves) {
    subsurface = site.hasCaves ? 98 : 10;
  } else {
    subsurface = site.hasCaves ? 65 : 85;
  }

  return {
    thermal: Math.round(thermal),
    radiation: Math.round(radiation),
    aridity: Math.round(aridity),
    topography: Math.round(topography),
    geology: Math.round(geology),
    subsurface: Math.round(subsurface)
  };
}

function calculateSimilarity(targetProfile, earthProfile, weights) {
  const dimensions = ['thermal', 'radiation', 'aridity', 'topography', 'geology', 'subsurface'];
  let weightedErrorSum = 0;
  let totalWeight = 0;

  for (const dim of dimensions) {
    const w = typeof weights[dim] === 'number' ? Math.max(0, weights[dim]) : 50;
    const diff = Math.abs(targetProfile[dim] - earthProfile[dim]);
    weightedErrorSum += w * diff;
    totalWeight += w;
  }

  if (totalWeight === 0) {
    totalWeight = dimensions.length;
    weightedErrorSum = dimensions.reduce(
      (acc, dim) => acc + Math.abs(targetProfile[dim] - earthProfile[dim]),
      0
    );
  }

  const meanWeightedDiff = weightedErrorSum / totalWeight;
  return Number(clamp(100 - meanWeightedDiff, 0, 100).toFixed(1));
}

function translatePlaceCategory(rawType = '') {
  const t = String(rawType).toLowerCase();
  if (t.includes('volcano') || t.includes('peak') || t.includes('mountain')) return 'Formación volcánica / montaña';
  if (t.includes('desert') || t.includes('natural') || t.includes('park') || t.includes('reserve')) return 'Área natural';
  if (t.includes('city') || t.includes('town') || t.includes('municipality') || t.includes('locality')) return 'Ciudad / Municipio';
  if (t.includes('village') || t.includes('hamlet')) return 'Localidad';
  if (t.includes('university') || t.includes('college') || t.includes('school')) return 'Institución';
  if (t.includes('admin') || t.includes('state') || t.includes('region')) return 'Región / Estado';
  return 'Ubicación geográfica';
}

app.get('/api/geocode', async (req, res) => {
  const q = req.query.q;
  if (!q || typeof q !== 'string' || q.trim().length < 2) {
    return res.status(400).json({ error: 'Escribe al menos 2 letras del lugar que deseas buscar.' });
  }

  const rawQuery = q.trim();

  try {
    // 3 geocodificadores para máxima velocidad y precisión
    const [arcgisRes, openMeteoRes, photonRes] = await Promise.allSettled([
      axios.get('https://geocode.arcgis.com/arcgis/rest/services/World/GeocodeServer/findAddressCandidates', {
        params: {
          SingleLine: rawQuery,
          outFields: 'Match_addr,PlaceName,Region,Country,Type',
          maxLocations: 7,
          langCode: 'ES',
          f: 'json'
        },
        timeout: 4500
      }),
      axios.get('https://geocoding-api.open-meteo.com/v1/search', {
        params: {
          name: rawQuery,
          count: 7,
          language: 'es',
          format: 'json'
        },
        timeout: 4500
      }),
      axios.get('https://photon.komoot.io/api/', {
        params: {
          q: rawQuery,
          limit: 6
        },
        timeout: 4500
      })
    ]);

    const candidates = [];

    const addCandidate = (name, subtitle, lat, lon, rawType) => {
      if (typeof lat !== 'number' || typeof lon !== 'number' || Number.isNaN(lat) || Number.isNaN(lon)) {
        return;
      }
      const cleanLat = Number(lat.toFixed(5));
      const cleanLon = Number(lon.toFixed(5));
      const isDuplicate = candidates.some(
        (c) => Math.abs(c.lat - cleanLat) < 0.02 && Math.abs(c.lon - cleanLon) < 0.02
      );
      if (isDuplicate) return;

      const cleanName = (name || rawQuery).trim();
      const cleanSub = (subtitle || '').trim();
      const fullName = cleanSub && !cleanName.includes(cleanSub)
        ? `${cleanName}, ${cleanSub}`
        : cleanName;

      candidates.push({
        name: cleanName,
        subtitle: cleanSub || 'Coordenada global',
        fullName,
        lat: cleanLat,
        lon: cleanLon,
        type: translatePlaceCategory(rawType)
      });
    };

    if (arcgisRes.status === 'fulfilled' && Array.isArray(arcgisRes.value.data?.candidates)) {
      for (const item of arcgisRes.value.data.candidates) {
        const attr = item.attributes || {};
        const loc = item.location || {};
        const placeName = attr.PlaceName || (attr.Match_addr ? attr.Match_addr.split(',')[0] : rawQuery);
        const subParts = [attr.Region, attr.Country].filter(
          (v, i, arr) => v && v !== placeName && arr.indexOf(v) === i
        );
        addCandidate(placeName, subParts.join(', '), loc.y, loc.x, attr.Type);
      }
    }

    if (openMeteoRes.status === 'fulfilled' && Array.isArray(openMeteoRes.value.data?.results)) {
      for (const item of openMeteoRes.value.data.results) {
        const subParts = [item.admin1, item.country].filter(
          (v, i, arr) => v && v !== item.name && arr.indexOf(v) === i
        );
        addCandidate(item.name, subParts.join(', '), item.latitude, item.longitude, item.feature_code);
      }
    }

    if (photonRes.status === 'fulfilled' && Array.isArray(photonRes.value.data?.features)) {
      for (const feat of photonRes.value.data.features) {
        const props = feat.properties || {};
        const coords = feat.geometry?.coordinates || [];
        if (coords.length < 2) continue;
        const primary = props.name || props.city || props.state || rawQuery;
        const subParts = [props.city !== primary ? props.city : '', props.state, props.country].filter(
          (v, i, arr) => v && v !== primary && arr.indexOf(v) === i
        );
        addCandidate(primary, subParts.join(', '), Number(coords[1]), Number(coords[0]), props.osm_value);
      }
    }

    res.json({
      query: rawQuery,
      results: candidates.slice(0, 8)
    });
  } catch (err) {
    console.error('Error en /api/geocode:', err.message);
    res.status(500).json({ error: 'No se pudo consultar el servicio de ubicaciones.' });
  }
});

app.post('/api/match', async (req, res) => {
  try {
    const { targetId, weights = {} } = req.body;
    const target = spaceTargets.find((t) => t.id === targetId) || spaceTargets[0];

    const evaluatedSites = await Promise.all(
      earthCandidates.map(async (site) => {
        const climate = await fetchNasaPowerByCoords(site.lat, site.lon, site.id, site.fallbackClimate);
        const earthProfile = buildEarthNormalizedProfile(site, climate, target);
        const planetarySimilarity = calculateSimilarity(target.profile, earthProfile, weights);

        const missionReadinessScore = Number(
          (planetarySimilarity * 0.65 + site.logisticViability * 0.35).toFixed(1)
        );

        return {
          id: site.id,
          name: site.name,
          country: site.country,
          category: site.category || 'understudied',
          isCustom: false,
          lat: site.lat,
          lon: site.lon,
          geologyType: site.geologyType,
          hasCaves: site.hasCaves,
          highlights: site.highlights,
          logisticNotes: site.logisticNotes,
          climateData: climate,
          scores: {
            planetarySimilarity,
            logisticViability: site.logisticViability,
            missionReadinessScore
          },
          critique: generateScientificCritique(target, earthProfile, climate),
          radarComparison: {
            labels: ['Frío extremo', 'Luz solar', 'Sequedad', 'Relieve', 'Tipo de roca', 'Cuevas'],
            targetValues: [
              target.profile.thermal,
              target.profile.radiation,
              target.profile.aridity,
              target.profile.topography,
              target.profile.geology,
              target.profile.subsurface
            ],
            earthValues: [
              earthProfile.thermal,
              earthProfile.radiation,
              earthProfile.aridity,
              earthProfile.topography,
              earthProfile.geology,
              earthProfile.subsurface
            ]
          }
        };
      })
    );

    evaluatedSites.sort((a, b) => b.scores.planetarySimilarity - a.scores.planetarySimilarity);

    res.json({
      target,
      totalAnalyzed: evaluatedSites.length,
      results: evaluatedSites
    });
  } catch (error) {
    console.error('Error en /api/match:', error);
    res.status(500).json({ error: 'Error al procesar el análisis' });
  }
});

      weights = {},
      geologyType = 'basaltic-volcanic',
      hasCaves = false,
      preferredName = ''
    } = req.body;

    if (typeof lat !== 'number' || typeof lon !== 'number') {
      return res.status(400).json({ error: 'Coordenadas inválidas' });
    }

    const target = spaceTargets.find((t) => t.id === targetId) || spaceTargets[0];

    const [climate, elevation, geoInfo] = await Promise.all([
      fetchNasaPowerByCoords(lat, lon),
      fetchElevationByCoords(lat, lon),
      reverseGeocodeCoords(lat, lon)
    ]);

    const finalPlaceName =
      typeof preferredName === 'string' && preferredName.trim().length > 0
        ? preferredName.trim()
        : geoInfo.placeName;

    const estimatedSlope =
      elevation !== null ? clamp(Math.round(25 + (Math.abs(elevation) / 4500) * 65), 20, 90) : 55;

    const customSite = {
      id: `custom-${lat.toFixed(4)}-${lon.toFixed(4)}`,
      name: finalPlaceName,
      country:
        elevation !== null
          ? `${geoInfo.regionSubtitle} · Altitud: ${elevation} m`
          : geoInfo.regionSubtitle,
      category: 'custom',
      isCustom: true,
      lat: Number(lat.toFixed(5)),
      lon: Number(lon.toFixed(5)),
      geologyType,
      hasCaves,
      slopeIndex: estimatedSlope,
      logisticViability: Math.abs(lat) > 65 ? 30 : 84
    };

    const earthProfile = buildEarthNormalizedProfile(customSite, climate, target);
    const planetarySimilarity = calculateSimilarity(target.profile, earthProfile, weights);
    const missionReadinessScore = Number(
      (planetarySimilarity * 0.65 + customSite.logisticViability * 0.35).toFixed(1)
    );

    res.json({
      ...customSite,
      highlights: `Ubicación exacta en ${finalPlaceName} (${lat.toFixed(4)}°, ${lon.toFixed(
        4
      )}°). Altitud aproximada: ${elevation !== null ? elevation + ' m s.n.m.' : 'no disponible'}.`,
      logisticNotes:
        Math.abs(lat) > 65
          ? 'Zona polar remota: requiere logística especializada.'
          : 'Ubicación accesible para estudios de campo y validación de instrumentos.',
      climateData: climate,
      scores: {
        planetarySimilarity,
        logisticViability: customSite.logisticViability,
        missionReadinessScore
      },
      critique: generateScientificCritique(target, earthProfile, climate),
      radarComparison: {
        labels: ['Frío extremo', 'Luz solar', 'Sequedad', 'Relieve', 'Tipo de roca', 'Cuevas'],
        targetValues: [
          target.profile.thermal,
          target.profile.radiation,
          target.profile.aridity,
          target.profile.topography,
          target.profile.geology,
          target.profile.subsurface
        ],
        earthValues: [
          earthProfile.thermal,
          earthProfile.radiation,
          earthProfile.aridity,
          earthProfile.topography,
          earthProfile.geology,
          earthProfile.subsurface
        ]
      }
    });
  } catch (error) {
    console.error('Error en /api/prospect:', error);
    res.status(500).json({ error: 'Error al explorar la coordenada' });
  }
});

app.post('/api/discoveries', (req, res) => {
  const { site, targetName, customName, notes, user } = req.body;
  if (!site) return res.status(400).json({ error: 'Faltan datos del sitio' });

  const db = readDB();
  const finalName = (customName && customName.trim()) || site.name;

  const newRecord = {
    ...site,
    id: `db-${Date.now()}`,
    userId: user?.id || 'user-demo-1',
    name: finalName,
    highlights: (notes && notes.trim()) || site.highlights,
    savedAt: new Date().toLocaleDateString('es-MX', {
      day: '2-digit',
      month: 'short',
      year: 'numeric'
    }),
    matchedTargetName: targetName || 'Misión espacial',
    authorName: user?.name || 'Investigador invitado',
    authorInstitution: user?.institution || 'Terra-Twin Lab'
  };

  db.discoveries.unshift(newRecord);
  writeDB(db);
  res.json(newRecord);
});

  constructor(
    private analogService: AnalogService,
    private cdr: ChangeDetectorRef
  ) { }

  get filteredResults(): EarthMatchResult[] {
    if (this.activeFilter === 'understudied') {
      return this.results.filter((s) => s.category === 'understudied');
    }
    if (this.activeFilter === 'caves') {
      return this.results.filter((s) => s.hasCaves);
    }
    return this.results;
  }

  ngOnInit(): void {
    const storedUser = localStorage.getItem('terratwin_user');
    if (storedUser) {
      try {
        this.currentUser = JSON.parse(storedUser);
        if (this.currentUser) {
          this.enterPlatformWorkspace();
        }
      } catch {
        this.currentUser = null;
      }
    }
  }

  // ACCESO A LA PLATAFORMA
  switchAuthMode(mode: 'login' | 'register'): void {
    this.authMode = mode;
    this.authError = '';
    this.cdr.detectChanges();
  }

  submitAuth(): void {
    this.authError = '';
    this.isAuthenticating = true;
    this.cdr.detectChanges();

    if (this.authMode === 'login') {
      this.analogService.login(this.authEmail, this.authPassword).subscribe({
        next: (user) => {
          this.isAuthenticating = false;
          this.completeSignIn(user);
        },
        error: (err) => {
          this.isAuthenticating = false;
          this.authError =
            err.error?.error || 'No se pudo iniciar sesión. Verifica que el servidor esté encendido.';
          this.cdr.detectChanges();
        }
      });
    } else {
      this.analogService
        .register(this.authName, this.authEmail, this.authPassword, this.authInstitution)
        .subscribe({
          next: (user) => {
            this.isAuthenticating = false;
            this.completeSignIn(user);
          },
          error: (err) => {
            this.isAuthenticating = false;
            this.authError =
              err.error?.error || 'No se pudo crear la cuenta. Verifica que el servidor esté encendido.';
            this.cdr.detectChanges();
          }
        });
    }
  }

  enterDemoMode(): void {
    this.authError = '';
    this.isAuthenticating = true;
    this.cdr.detectChanges();

    this.analogService.loginDemo().subscribe({
      next: (user) => {
        this.isAuthenticating = false;
        this.completeSignIn(user);
      },
      error: () => {
        this.isAuthenticating = false;
        this.authError = 'No se detecta el servidor en http://localhost:3000. Enciéndelo primero.';
        this.cdr.detectChanges();
      }
    });
  }

  private completeSignIn(user: UserProfile): void {
    this.currentUser = user;
    localStorage.setItem('terratwin_user', JSON.stringify(user));
    this.activeTab = 'explorer';
    this.cdr.detectChanges();
    this.enterPlatformWorkspace();
  }

  private enterPlatformWorkspace(): void {
    this.loadInitialData();
    this.loadSavedDiscoveries();
    setTimeout(() => {
      this.initMap();
    }, 120);
  }

  logout(): void {
    this.currentUser = null;
    localStorage.removeItem('terratwin_user');
    if (this.map) {
      this.map.remove();
      this.map = null;
    }
    this.destroyOrbitalMaps();
    if (this.radarChart) {
      this.radarChart.destroy();
      this.radarChart = null;
    }
    this.cdr.detectChanges();
  }

  // NAVEGACIÓN Y MAPAS
  setTab(tab: TabId): void {
    if (this.activeTab === 'analysis' && tab !== 'analysis') {
      this.destroyOrbitalMaps();
    }

    this.activeTab = tab;
    this.cdr.detectChanges();

    if (tab === 'explorer') {
      setTimeout(() => {
        if (!this.map) {
          this.initMap();
        } else {
          this.map.invalidateSize();
        }
      }, 80);
    } else if (tab === 'analysis') {
      setTimeout(() => {
        this.initOrbitalMaps();
        this.renderRadarChart();
      }, 90);
    } else if (tab === 'database') {
      this.loadSavedDiscoveries();
    }
  }

  private initMap(): void {
    if (!this.mapContainer || this.map) return;

    const el = this.mapContainer.nativeElement;

    this.map = L.map(el, {
      center: [24.0277, -104.6532],
      zoom: 4,
      zoomControl: true
    });

    L.tileLayer(
      'https://server.arcgisonline.com/ArcGIS/rest/services/World_Imagery/MapServer/tile/{z}/{y}/{x}',
      {
        attribution: 'NASA / Esri World Imagery',
        maxZoom: 18
      }
    ).addTo(this.map);

    this.nasaGibsLayer = L.tileLayer(
      'https://gibs.earthdata.nasa.gov/wmts/epsg3857/best/MODIS_Terra_CorrectedReflectance_TrueColor/default/2026-10-15/GoogleMapsCompatible_Level9/{z}/{y}/{x}.jpg',
      {
        attribution: 'NASA EOSDIS GIBS',
        maxZoom: 9,
        opacity: 0.78
      }
    );

    L.tileLayer(
      'https://{s}.basemaps.cartocdn.com/dark_only_labels/{z}/{x}/{y}{r}.png',
      {
        subdomains: 'abcd',
        maxZoom: 18
      }
    ).addTo(this.map);

    this.markersLayer.addTo(this.map);

    this.map.on('click', (e: L.LeafletMouseEvent) => {
      this.searchSuggestions = [];
      this.searchDropdownOpen = false;
      if (this.prospectMode) {
        this.analyzeCustomCoordinate(e.latlng.lat, e.latlng.lng, '', 10);
      }
    });

    setTimeout(() => {
      this.map?.invalidateSize();
    }, 250);

    const observer = new ResizeObserver(() => {
      this.map?.invalidateSize();
    });
    observer.observe(el);

    if (this.results.length > 0) {
      this.updateMapMarkers();
    }
  }

  // VISOR ORBITAL DE LA LUNA Y MARTE
  private parseTargetCoordinates(coordStr: string): { lat: number; lon: number } {
    const match = coordStr.match(/([\d.]+)\s*°?\s*([NS])\s*,\s*([\d.]+)\s*°?\s*([EW])/i);
    if (!match) {
      return { lat: 0, lon: 0 };
    }
    let lat = parseFloat(match[1]);
    if (match[2].toUpperCase() === 'S') lat = -lat;

    let lon = parseFloat(match[3]);
    if (match[4].toUpperCase() === 'W') lon = -lon;
    if (lon > 180) lon = lon - 360;

    const safeLat = Math.max(-80, Math.min(80, lat));
    return { lat: safeLat, lon };
  }

  private initOrbitalMaps(): void {
    this.destroyOrbitalMaps();

    if (!this.activeTarget || !this.selectedSite) return;

    if (this.planetaryMapContainer) {
      const planetEl = this.planetaryMapContainer.nativeElement;
      const coords = this.parseTargetCoordinates(this.activeTarget.coordinates);
      const isMars = this.activeTarget.celestialBody === 'Marte';

      const trekUrl = isMars
        ? 'https://trek.nasa.gov/tiles/Mars/EQ/Mars_MGS_MOLA_ClrShade_merge_global_463m/1.0.0/default/default028mm/{z}/{y}/{x}.jpg'
        : 'https://trek.nasa.gov/tiles/Moon/EQ/LRO_WAC_Mosaic_Global_303ppd_v02/1.0.0/default/default028mm/{z}/{y}/{x}.jpg';

      this.planetaryMap = L.map(planetEl, {
        crs: L.CRS.EPSG4326,
        center: [coords.lat, coords.lon],
        zoom: 4,
        minZoom: 1,
        maxZoom: 7,
        maxBounds: [
          [-90, -180],
          [90, 180]
        ],
        zoomControl: true
      });

      L.tileLayer(trekUrl, {
        attribution: isMars ? 'NASA Mars Trek · MGS MOLA' : 'NASA Moon Trek · LRO WAC',
        noWrap: true,
        bounds: [
          [-90, -180],
          [90, 180]
        ]
      }).addTo(this.planetaryMap);

      const targetIcon = L.divIcon({
        className: 'nasa-pin-wrapper',
        html: `<div class="orbital-reticle"></div>`,
        iconSize: [28, 28],
        iconAnchor: [14, 14]
      });

      L.marker([coords.lat, coords.lon], { icon: targetIcon })
        .bindTooltip(
          `<strong>${this.activeTarget.name}</strong><br/>${this.activeTarget.coordinates}`,
          { direction: 'top' }
        )
        .addTo(this.planetaryMap);

      setTimeout(() => {
        this.planetaryMap?.invalidateSize();
      }, 150);
    }

    if (this.earthCompareMapContainer) {
      const earthEl = this.earthCompareMapContainer.nativeElement;
      const { lat, lon, name, country } = this.selectedSite;

      this.earthCompareMap = L.map(earthEl, {
        center: [lat, lon],
        zoom: 12,
        zoomControl: true
      });

      L.tileLayer(
        'https://server.arcgisonline.com/ArcGIS/rest/services/World_Imagery/MapServer/tile/{z}/{y}/{x}',
        {
          attribution: 'NASA / Esri World Imagery',
          maxZoom: 18
        }
      ).addTo(this.earthCompareMap);

      const earthIcon = L.divIcon({
        className: 'nasa-pin-wrapper',
        html: `<div class="nasa-pin selected"></div>`,
        iconSize: [22, 22],
        iconAnchor: [11, 11]
      });

      L.marker([lat, lon], { icon: earthIcon })
        .bindTooltip(`<strong>${name}</strong><br/>${country}`, { direction: 'top' })
        .addTo(this.earthCompareMap);

      setTimeout(() => {
        this.earthCompareMap?.invalidateSize();
      }, 150);
    }
  }

  recenterPlanetaryOrbit(): void {
    if (!this.planetaryMap || !this.activeTarget) return;
    const coords = this.parseTargetCoordinates(this.activeTarget.coordinates);
    this.planetaryMap.flyTo([coords.lat, coords.lon], 4, { duration: 1 });
  }

  recenterEarthOrbit(): void {
    if (!this.earthCompareMap || !this.selectedSite) return;
    this.earthCompareMap.flyTo([this.selectedSite.lat, this.selectedSite.lon], 12, { duration: 1 });
  }

  private destroyOrbitalMaps(): void {
    if (this.planetaryMap) {
      this.planetaryMap.remove();
      this.planetaryMap = null;
    }
    if (this.earthCompareMap) {
      this.earthCompareMap.remove();
      this.earthCompareMap = null;
    }
  }

  toggleNasaGibs(): void {
    if (!this.map || !this.nasaGibsLayer) return;
    this.showNasaGibsLayer = !this.showNasaGibsLayer;
    if (this.showNasaGibsLayer) {
      this.nasaGibsLayer.addTo(this.map);
    } else {
      this.map.removeLayer(this.nasaGibsLayer);
    }
  }

  toggleProspectMode(): void {
    this.prospectMode = !this.prospectMode;
  }

  // BUSCADOR DE UBICACIONES
  onSearchInputChange(): void {
    if (this.searchTimer) {
      clearTimeout(this.searchTimer);
    }

    const q = this.searchQuery.trim();
    this.searchNoResults = false;

    if (q.length < 2) {
      this.searchSuggestions = [];
      this.searchDropdownOpen = false;
      return;
    }

    this.searchTimer = setTimeout(() => {
      this.fetchLocationOptions(q);
    }, 300);
  }

  searchLocationByText(): void {
    const q = this.searchQuery.trim();
    if (q.length < 2) return;

    if (this.searchTimer) {
      clearTimeout(this.searchTimer);
    }

    this.fetchLocationOptions(q);
  }

  private fetchLocationOptions(q: string): void {
    this.isSearchingPlace = true;
    this.searchDropdownOpen = true;
    this.searchNoResults = false;
    this.errorMessage = '';
    this.cdr.detectChanges();

    this.analogService.searchPlace(q).subscribe({
      next: (res) => {
        this.isSearchingPlace = false;
        this.searchSuggestions = res.results || [];
        this.searchNoResults = this.searchSuggestions.length === 0;
        this.searchDropdownOpen = true;
        this.cdr.detectChanges();
      },
      error: () => {
        this.isSearchingPlace = false;
        this.searchSuggestions = [];
        this.searchNoResults = true;
        this.searchDropdownOpen = true;
        this.cdr.detectChanges();
      }
    });
  }

  closeSearchDropdown(): void {
    this.searchDropdownOpen = false;
  }

  selectSearchSuggestion(item: GeocodeCandidate): void {
    const selectedLabel = item.fullName || item.name;
    this.searchQuery = selectedLabel;
    this.searchSuggestions = [];
    this.searchDropdownOpen = false;
    this.searchNoResults = false;
    this.prospectMode = true;
    this.analyzeCustomCoordinate(item.lat, item.lon, selectedLabel, 12);
  }

  analyzeCustomCoordinate(
    lat: number,
    lon: number,
    preferredName: string = '',
    zoomLevel: number = 11
  ): void {
    this.isProspecting = true;
    this.saveStatusMessage = '';
    this.cdr.detectChanges();

    this.analogService
      .prospectCoordinate(
        lat,
        lon,
        this.selectedTargetId,
        this.weights,
        this.prospectGeology,
        this.prospectHasCaves,
        preferredName
      )
      .subscribe({
        next: (discoveredSite) => {
          this.customPinSite = discoveredSite;
          this.customSiteName = discoveredSite.name;
          this.customSiteNotes = discoveredSite.highlights;
          this.isProspecting = false;
          this.selectSite(discoveredSite, true, false, zoomLevel);
          this.cdr.detectChanges();
        },
        error: () => {
          this.isProspecting = false;
          this.errorMessage = 'No se pudo consultar esa coordenada. Verifica que el servidor esté encendido.';
          this.cdr.detectChanges();
        }
      });
  }

  saveCurrentToDatabase(): void {
    if (!this.selectedSite) return;

    this.isSavingDB = true;
    this.saveStatusMessage = '';
    this.cdr.detectChanges();

    this.analogService
      .saveDiscovery(
        this.selectedSite,
        this.activeTarget?.name || 'Misión espacial',
        this.customSiteName,
        this.customSiteNotes,
        this.currentUser
      )
      .subscribe({
        next: (savedItem) => {
          this.isSavingDB = false;
          this.saveStatusMessage = `Guardado en tu base de datos (${savedItem.name})`;
          this.loadSavedDiscoveries();
          this.cdr.detectChanges();
        },
        error: () => {
          this.isSavingDB = false;
          this.saveStatusMessage = 'No se pudo guardar. Verifica que el servidor esté encendido.';
          this.cdr.detectChanges();
        }
      });
  }

  loadSavedDiscoveries(): void {
    this.analogService.getDiscoveries(this.currentUser?.id).subscribe({
      next: (list) => {
        this.savedDiscoveries = list;
        this.updateMapMarkers();
        this.cdr.detectChanges();
      }
    });
  }

  deleteFromDatabase(id: string): void {
    this.analogService.deleteDiscovery(id).subscribe({
      next: () => {
        this.loadSavedDiscoveries();
      }
    });
  }

  loadInitialData(): void {
    this.isLoading = true;
    this.analogService.getTargets().subscribe({
      next: (data) => {
        this.targets = data;
        if (data.length > 0) {
          this.selectedTargetId = data[0].id;
          this.runSimulation();
        }
        this.cdr.detectChanges();
      },
      error: () => {
        this.errorMessage =
          'No se detecta el servidor en http://localhost:3000. Enciéndelo con node server.js.';
        this.isLoading = false;
        this.cdr.detectChanges();
      }
    });
  }

  onTargetChange(): void {
    const target = this.targets.find((t) => t.id === this.selectedTargetId);
    if (target) {
      this.weights.subsurface = target.requiresCaves ? 100 : 25;
    }
    this.runSimulation();
  }

  runSimulation(): void {
    this.isLoading = true;
    this.errorMessage = '';

    this.analogService.findMatches(this.selectedTargetId, this.weights).subscribe({
      next: (response) => {
        this.activeTarget = response.target;
        this.results = response.results;
        this.isLoading = false;

        if (this.customPinSite && this.selectedSite?.id === this.customPinSite.id) {
          this.analyzeCustomCoordinate(
            this.customPinSite.lat,
            this.customPinSite.lon,
            this.customPinSite.name,
            12
          );
          return;
        }

        if (this.results.length > 0) {
          const stillExists = this.results.find((r) => r.id === this.selectedSite?.id);
          this.selectSite(stillExists || this.results[0], false);
        }

        this.updateMapMarkers();

        if (this.activeTab === 'analysis') {
          setTimeout(() => {
            this.initOrbitalMaps();
            this.renderRadarChart();
          }, 80);
        }

        this.cdr.detectChanges();
      },
      error: () => {
        this.errorMessage = 'Error al calcular similitudes.';
        this.isLoading = false;
        this.cdr.detectChanges();
      }
    });
  }

  selectSite(
    site: EarthMatchResult,
    flyToLocation: boolean = true,
    switchToExplorer: boolean = false,
    zoomLevel: number = 10
  ): void {
    this.selectedSite = site;
    this.customSiteName = site.name;
    this.customSiteNotes = site.highlights;
    this.saveStatusMessage = '';

    if (switchToExplorer) {
      if (this.activeTab === 'analysis') {
        this.destroyOrbitalMaps();
      }
      this.activeTab = 'explorer';
    }

    this.updateMapMarkers();
    this.cdr.detectChanges();

    if (flyToLocation && this.map && this.activeTab === 'explorer') {
      setTimeout(() => {
        this.map?.invalidateSize();
        this.map?.flyTo([site.lat, site.lon], zoomLevel, { duration: 1.1 });
      }, 60);
    }

    if (this.activeTab === 'analysis') {
      setTimeout(() => {
        this.initOrbitalMaps();
        this.renderRadarChart();
      }, 80);
    }
  }

  exportMissionReport(): void {
    window.print();
  }

  private updateMapMarkers(): void {
    if (!this.map) return;
    this.markersLayer.clearLayers();

    this.results.forEach((site, index) => {
      const isSelected = this.selectedSite?.id === site.id;
      const isTop3 = index < 3;
      const pinClass = `nasa-pin ${isTop3 ? 'top-match' : ''} ${isSelected ? 'selected' : ''}`;

      const icon = L.divIcon({
        className: 'nasa-pin-wrapper',
        html: `<div class="${pinClass}"></div>`,
        iconSize: [20, 20],
        iconAnchor: [10, 10]
      });

      const marker = L.marker([site.lat, site.lon], { icon });
      marker.bindTooltip(
        `<strong>${site.name}</strong><br/>Parecido: ${site.scores.planetarySimilarity}%`,
        { direction: 'top' }
      );
      marker.on('click', () => {
        this.selectSite(site, true, false, 10);
      });
      marker.addTo(this.markersLayer);
    });

    const extraPins = [...this.savedDiscoveries];
    if (this.customPinSite && !extraPins.some((p) => p.id === this.customPinSite?.id)) {
      extraPins.push(this.customPinSite);
    }

    extraPins.forEach((site) => {
      const isSelected = this.selectedSite?.id === site.id;
      const icon = L.divIcon({
        className: 'nasa-pin-wrapper',
        html: `<div class="nasa-pin custom-pin ${isSelected ? 'selected' : ''}"></div>`,
        iconSize: [22, 22],
        iconAnchor: [11, 11]
      });
      const marker = L.marker([site.lat, site.lon], { icon });
      marker.bindTooltip(`<strong>${site.name}</strong>`, { direction: 'top' });
      marker.on('click', () => {
        this.selectSite(site, true, false, 12);
      });
      marker.addTo(this.markersLayer);
    });
  }

  private renderRadarChart(): void {
    if (!this.radarCanvas || !this.selectedSite || !this.activeTarget) return;
    const ctx = this.radarCanvas.nativeElement.getContext('2d');
    if (!ctx) return;

    if (this.radarChart) {
      this.radarChart.destroy();
    }

    const { labels, targetValues, earthValues } = this.selectedSite.radarComparison;

    this.radarChart = new Chart(ctx, {
      type: 'radar',
      data: {
        labels,
        datasets: [
          {
            label: `${this.activeTarget.name} (${this.activeTarget.celestialBody})`,
            data: targetValues,
            backgroundColor: 'rgba(225, 29, 72, 0.22)',
            borderColor: '#ff2a4b',
            pointBackgroundColor: '#ff2a4b',
            borderWidth: 2.5
          },
          {
            label: `${this.selectedSite.name} (Tierra)`,
            data: earthValues,
            backgroundColor: 'rgba(56, 189, 248, 0.22)',
            borderColor: '#38bdf8',
            pointBackgroundColor: '#38bdf8',
            borderWidth: 2.5
          }
        ]
      },
      options: {
        responsive: true,
        maintainAspectRatio: false,
        scales: {
          r: {
            min: 0,
            max: 100,
            ticks: { display: false, stepSize: 25 },
            grid: { color: 'rgba(255, 255, 255, 0.12)' },
            angleLines: { color: 'rgba(255, 255, 255, 0.12)' },
            pointLabels: {
              color: '#e4e4e7',
              font: { size: 12, weight: 'bold', family: 'DM Sans' }
            }
          }
        },
        plugins: {
          legend: {
            position: 'bottom',
            labels: {
              color: '#f4f4f6',
              padding: 16,
              font: { size: 12, family: 'DM Sans' }
            }
          }
        }
      }
    });
  }
}

@Injectable({
  providedIn: 'root'
})
export class AnalogService {
  private readonly apiUrl = 'http://localhost:3000/api';

  constructor(private http: HttpClient) { }

  login(email: string, password: string): Observable<UserProfile> {
    return this.http.post<UserProfile>(`${this.apiUrl}/auth/login`, { email, password });
  }

  register(name: string, email: string, password: string, institution: string): Observable<UserProfile> {
    return this.http.post<UserProfile>(`${this.apiUrl}/auth/register`, {
      name,
      email,
      password,
      institution
    });
  }

  loginDemo(): Observable<UserProfile> {
    return this.http.post<UserProfile>(`${this.apiUrl}/auth/demo`, {});
  }

  searchPlace(query: string): Observable<GeocodeResponse> {
    return this.http.get<GeocodeResponse>(`${this.apiUrl}/geocode`, {
      params: { q: query }
    });
  }

  getTargets(): Observable<SpaceTarget[]> {
    return this.http.get<SpaceTarget[]>(`${this.apiUrl}/targets`);
  }

  findMatches(targetId: string, weights: EnvironmentalProfile): Observable<MatchResponse> {
    return this.http.post<MatchResponse>(`${this.apiUrl}/match`, {
      targetId,
      weights
    });
  }

  prospectCoordinate(
    lat: number,
    lon: number,
    targetId: string,
    weights: EnvironmentalProfile,
    geologyType: string,
    hasCaves: boolean,
    preferredName: string = ''
  ): Observable<EarthMatchResult> {
    return this.http.post<EarthMatchResult>(`${this.apiUrl}/prospect`, {
      lat,
      lon,
      targetId,
      weights,
      geologyType,
      hasCaves,
      preferredName
    });
  }

  getDiscoveries(userId?: string): Observable<EarthMatchResult[]> {
    const params: Record<string, string> = {};
    if (userId) {
      params['userId'] = userId;
    }
    return this.http.get<EarthMatchResult[]>(`${this.apiUrl}/discoveries`, { params });
  }

  saveDiscovery(
    site: EarthMatchResult,
    targetName: string,
    customName: string,
    notes: string,
    user: UserProfile | null
  ): Observable<EarthMatchResult> {
    return this.http.post<EarthMatchResult>(`${this.apiUrl}/discoveries`, {
      site,
      targetName,
      customName,
      notes,
      user
    });
  }

  deleteDiscovery(id: string): Observable<{ deleted: boolean }> {
    return this.http.delete<{ deleted: boolean }>(`${this.apiUrl}/discoveries/${id}`);
  }
}
